    \posterfig{\linewidth}{tikz-radar-lib/problem_panel_strip.pdf}{Radar point clouds are sparse, noisy, and contaminated by multipath ghosts}

\normalsize
\begin{tabular}{@{}c p{0.85\linewidth}@{}}
    \includegraphics[width=1.3cm]{icon_sparse.pdf} & \textbf{Low Angular Resolution} \\
    & Conventional angle-of-arrival~\cite{ZhangMUSIC2010} yields motion-defocused clouds and multipath ghost artifacts. \\[4mm]
    \includegraphics[width=1.3cm]{icon_rigid.pdf} & \textbf{Rigid-Target Bias} \\
    & Interferometric Inverse Synthetic Aperture Radar (InISAR) improves resolution but is validated only on large \emph{rigid} targets~\cite{QunZhangInISAR2004, Martorella2DISARto3D2013}. \\[4mm]
    \includegraphics[width=2.3cm]{icon_multipath.pdf} & \textbf{Multipath Ghosts} \\
    & Ghosts overlap in range-Doppler and mimic real 3D clusters~\cite{RuoyuMultipath}; indistinguishable for DBSCAN. \\[4mm]
    \includegraphics[width=1.1cm]{icon_scarcity.pdf} & \textbf{Data Scarcity} \\
    & Existing SOTA~\cite{Ma23, Pegoraro21} demand \(>100\) sequences/person; Point Cloud Transformer (PCT)~\cite{Guo20} handles spatial structure but targets \emph{static} objects~\cite{meucci23}. \\
\end{tabular}}

    \block{\textcolor{white}{\large \ding{206}\ \ InISAR Pipeline: 3D Reconstruction}}{
    
    \vspace{4mm}

    \normalsize
    \begin{minipage}{\linewidth}
    
        \vspace{4mm}
    
        \textbf{Step 1: Range FFT}
        \begin{itemize}
            \setlength\itemsep{0.5mm}
            \item FMCW chirp compression to range profile \(S_{R,m}(f,t)\). Physical range limit \(\Delta r = c/(2B)\) bounds our achieved height error.
        \end{itemize}
        \vspace{2mm}
        \textbf{Step 2: Motion Compensation (ICBA)}
        \begin{itemize}
            \setlength\itemsep{0.5mm}
            \item Translational motion model~\cite{Martorella2DISARto3D2013, Javadi_ISAR}:
        \end{itemize}
        \begin{align*}
            R_0(t) = r_0 + \beta t + \gamma t^2
        \end{align*}
        \begin{itemize}
            \setlength\itemsep{0.5mm}
            \item Nelder-Mead optimizes image contrast \([\hat{\beta}, \hat{\gamma}]\) to estimate global translation, correcting smearing caused by human motion.
        \end{itemize}
        \begin{align*}
            [\hat{\beta}, \hat{\gamma}] = \arg\max \left\{ \frac{\sqrt{\frac{1}{N}\sum(I_i - \bar{I})^2}}{\bar{I}} \right\}
        \end{align*}
        \begin{itemize}
            \setlength\itemsep{0.5mm}
            \item Yields signal-to-noise ratio (SNR) enhanced range profile $S_{C,m}(f,t)$.
        \end{itemize}
        \vspace{2mm}
        \textbf{Step 3:  Coherent Combination \& Doppler FFT}
        \begin{itemize}
            \setlength\itemsep{0.5mm}
            \item Geometric phase corrections align all MIMO virtual antennas (SNR boost) \(\to\) focused 2D ISAR Range-Doppler Profiles (RDP).
        \end{itemize}
        \vspace{2mm}
        \textbf{Step 4: CFAR Segmentation}
        \begin{itemize}
            \setlength\itemsep{0.5mm}
            \item SOCA-CFAR ($N_T=6$, $P_{FA}=10^{-10}$) isolates the target cells from noise/clutter.
        \end{itemize}
        \vspace{2mm}
        \textbf{Step 5: Interferometry}
        \begin{itemize}
            \setlength\itemsep{0.5mm}
            \item Phase difference between coherently combined top/bottom MIMO rows yields elevation:
        \end{itemize}
        \begin{align*}
            I_1 \cdot I_2^* = |I_1||I_2| \exp\left(j\frac{2\pi}{\lambda}\Delta R\right)
        \end{align*}
        \begin{align*}
            \phi = \frac{2\pi}{\lambda} L\frac{h}{R}, \quad h = \frac{\lambda \phi R}{2\pi L}
        \end{align*}
        \begin{itemize}
            \setlength\itemsep{0.5mm}
            \item Exploiting MIMO antenna pairs reduces measurement noise.
        \end{itemize}
        \vspace{2mm}
        \textbf{Step 6: EKF-JPDA Tracking}
        \begin{itemize}
            \setlength\itemsep{0.5mm}
            \item JPDA handles data association; EKF filters state covariance.
            \item Max miss count=1 prunes multipath ghosts while tolerating occlusions.
        \end{itemize}

\subcolumn{0.5}
    \block{\textcolor{white}{\large \ding{207}\ \ Few-Shot PID with PCTtrans}}{
    \normalsize
    \posterfig{\linewidth}{fig_pcttrans_architecture.pdf}{PCTtrans (PCT-T): frame-wise PCT $+$ Temporal Transformer}
    \begin{minipage}{\linewidth}
    \textbf{Postprocessing:}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item DBSCAN ($k=8$, $S=1.0$) + near-field filter
        \item Centroid normalization: $p_i' = p_i - (\bar{x}, \bar{y}, 0)$
        \item Random padding to $n_{\max}=1000$
        \item Output: $\mathcal{S} = \{\mathcal{P}_t\}_{t=1}^{10}$, $\mathcal{P}_t \in \mathbb{R}^{1000 \times 5}$
    \end{itemize}
    \textbf{Spatial (PCT Encoder~\cite{Guo20}):}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item Sampling-and-Grouping (SG): SG1: Farthest Point Sampling (FPS) $n_1=200$, $r_1=0.2$; SG2: FPS $n_2=100$, $r_2=0.3$
        \item Neighbor Embedding: $\Delta F(p) = \text{concat}_{q \in \text{knn}(p,k)} (F(q) - F(p))$
        \item Offset-Attention: $F_{\text{out}} = \text{LBR}(F_{\text{in}} - F_{\text{sa}}) + F_{\text{in}}$
        \item Output: $\mathbf{f}_t \in \mathbb{R}^{192}$
    \end{itemize}
    \textbf{Temporal (Transformer):}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item Positional encoding: $\mathbf{f}_t^{\text{pos}} = \mathbf{f}_t + \text{PE}_t$
        \item Multi-head self-attention ($h=4$)
        \item Max + average pooling $\to$ $\mathbf{g} \in \mathbb{R}^{192}$
        \item Classification:
        $$p(y|\mathcal{S}) = \text{Softmax}(W_2 \cdot \text{ReLU}(W_1 \mathbf{g} + b_1) + b_2)$$
    \end{itemize}
    \textbf{Key Design Choices:}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item FPS + Ball Query handles non-uniform density.
        \item Offset-Attention emphasizes local structure.
        \item Temporal transformer models gait dynamics.
    \end{itemize}
    \end{minipage}
    }
    \block{\textcolor{white}{\large \ding{209}\ \ Pre-trained Meta-Learning~\cite{Lin20}}}{
        \normalsize
        \centering
        \textbf{Two-Phase Algorithm:}\\
        \vspace{1mm}
        \textbf{Phase 1:} Supervised training on 3 subjects $\to \theta_0^{PT}$ \\
        \textbf{Phase 2:} Model-Agnostic Meta-Learning (MAML)~\cite{finn17, Puri20} meta-training with $\theta_0^{PT}$\\
        \vspace{2mm}
        Inner loop: $\theta_i' = \theta - \alpha\nabla\mathcal{L}$ ($\alpha{=}0.1$). Outer loop: $\theta \leftarrow \theta - \beta\nabla\sum_i\mathcal{L}$ ($\beta{=}0.001$).
        \vspace{2mm}
        \centering
        \colorbox{resultsColor!12}{\parbox{0.95\linewidth}{\centering
            \textbf{Gain:} +15.6pp over random MAML
        }}
