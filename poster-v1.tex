    \small
    \posterfig{\linewidth}{tikz-radar-lib/problem_panel_strip.pdf}{Radar point clouds are sparse, noisy, and contaminated by multipath ghosts}

\normalsize
\begin{tabular}{@{}c p{0.82\linewidth}@{}}
    \includegraphics[width=1.3cm]{icon_sparse.pdf} & \textbf{Low Angular Resolution} \\
    & Conventional angle-of-arrival~\cite{ZhangMUSIC2010} yields motion-defocused clouds and multipath artifacts. \\[4mm]
    \includegraphics[width=1.3cm]{icon_rigid.pdf} & \textbf{Rigid-Target Bias} \\
    & Interferometric Inverse Synthetic Aperture Radar (InISAR) improves resolution but is validated only on large \emph{rigid} targets~\cite{QunZhangInISAR2004, Martorella2DISARto3D2013}. \\[4mm]
    \includegraphics[width=2.3cm]{icon_multipath.pdf} & \textbf{Multipath Ghosts} \\
    & Ghosts overlap in range-Doppler and mimic real 3D clusters~\cite{RuoyuMultipath} \\[4mm]
    \includegraphics[width=1.1cm]{icon_scarcity.pdf} & \textbf{Data Scarcity} \\
    & Existing SOTA~\cite{Ma23, Pegoraro21} demand \(>100\) seq/person; Point Cloud Transformer (PCT)~\cite{Guo20} handles spatial structure but targets \emph{static} objects~\cite{meucci23}. \\
\end{tabular}}
