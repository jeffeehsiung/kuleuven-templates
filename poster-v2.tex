\block{\textcolor{white}{\large \ding{202}\ \ Introduction}}{
\begin{minipage}{\linewidth}
Vision-based person identification struggles under real conditions:
\begin{itemize}
  \setlength\itemsep{1.62mm}
  \item \textbf{Low illumination} degraded visual features
  \item \textbf{Adverse weather} fog and rain reduce reliability
  \item \textbf{Occlusions} missing body information
  \item \textbf{Privacy concerns} -- facial capture raises regulatory and ethical issues
\end{itemize}
\vspace{1.62mm}
mmWave radar provides a privacy-preserving alternative for surveillance and human-robot interaction, 
maintaining robustness without capturing identifiable imagery.
\end{minipage}

}

% ============================================================
% SECTION 2 -- CHALLENGES IN RADAR-BASED PID
% ============================================================
\block{\textcolor{white}{\large \ding{203}\ \ Challenges in Radar-Based PID}}{
\begin{minipage}{\linewidth}
    \small
    \posterfig{\linewidth}{tikz-radar-lib/problem_panel_strip.pdf}{Radar point clouds are sparse, noisy, and contaminated by multipath ghosts}
\end{minipage}
\begin{minipage}{\linewidth}
    \vspace{2mm}
    \textbf{Challenge 1: Poor Point Cloud Quality}\\[1mm]
    Conventional angle-of-arrival processing yields sparse, noisy point clouds with limited angular resolution, motion-induced defocusing, and multipath ghost artifacts.
    \vspace{2mm}
    
    \textbf{Challenge 2: Rigid-Target Bias in InISAR}\\[1mm]
    InISAR improves angular resolution but existing works validate it only on large \emph{rigid} targets -- \textbf{not} articulated human targets.
    \vspace{2mm}
    
    \textbf{Challenge 3: Multipath Cannot Be Ignored}\\[1mm]
    DBSCAN alone cannot distinguish true targets from multipath ghosts, which overlap in range-Doppler and mimic real 3D clusters.
    \vspace{2mm}
    
    \textbf{Challenge 4: Data Scarcity}\\[1mm]
    Existing methods need $>$100 sequences/person. PCT handles spatial structure but targets \emph{static} objects, not evolving radar sequences.
    \vspace{3mm}
\end{minipage}
\begin{minipage}{\linewidth}
    \centering
    \colorbox{resultsColor!12}{\parbox{0.95\linewidth}{\centering\textbf{Gap:}
    We bridge these gaps with tracking-integrated InISAR and pre-trained few-shot meta-learning.}}
\end{minipage}
}

\begin{subcolumns}

\subcolumn{0.5}
    \block{\textcolor{white}{\large \ding{206}\ \ InISAR Pipeline: 3D Reconstruction}}{
    \small
    \begin{minipage}{\linewidth}
    \vspace{0mm}
    \posterfig{\linewidth}{tikz-radar-lib/inisar_pipeline_strip_landscape.pdf}{Six-stage 3D reconstruction pipeline}
    \vspace{2mm}
    \textbf{Step 1: Range FFT}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item FMCW radar transmits linear frequency-modulated chirps:
    \end{itemize}
    \begin{align*}
        \Delta r = \frac{c}{2B} \quad \text{(range resolution)}
    \end{align*}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item Output: Range-compressed signal $S_{R,m}(f,t)$.
    \end{itemize}
    \vspace{1mm}
    \textbf{Step 2: ISAR Motion Compensation (ICBA)}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item Target motion during CPI smears energy across cells, defocusing the image.
        \item Translational motion model:
    \end{itemize}
    \begin{align*}
        R_0(t) = r_0 + \beta t + \gamma t^2
    \end{align*}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item ICBA maximizes image contrast to estimate $[\hat{\beta}, \hat{\gamma}]$:
    \end{itemize}
    \begin{align*}
        [\hat{\beta}, \hat{\gamma}] = \arg\max \left\{ \frac{\sqrt{\frac{1}{N}\sum(I_i - \bar{I})^2}}{\bar{I}} \right\}
    \end{align*}
    \vspace{1mm}
    \textbf{Step 3: Doppler FFT}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item Output: Range-Doppler map with focused target signatures.
    \end{itemize}
    \vspace{1mm}
    \textbf{Step 4: CFAR Segmentation}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item SOCA-CFAR ($N_T=6$, $P_{FA}=10^{-10}$) segments the target cells from noise/clutter.
    \end{itemize}
    \vspace{1mm}
    \textbf{Step 5: Interferometry}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item Uses phase difference between top and bottom antennas:
    \end{itemize}
    \begin{align*}
        I_1 \cdot I_2^* = |I_1||I_2| \exp\left(j\frac{2\pi}{\lambda}\Delta R\right)
    \end{align*}
    \begin{align*}
        \phi = \frac{2\pi}{\lambda} L\frac{h}{R}, \quad h = \frac{\lambda \phi R}{2\pi L}
    \end{align*}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item 3D coordinates:
    \end{itemize}
    \begin{align*}
        x = R\cos\theta_{el}\sin\theta_{az} \quad y = R\cos\theta_{el}\cos\theta_{az} \quad z = R\sin\theta_{el}
    \end{align*}
    \vspace{1mm}
    \textbf{Step 6: EKF-JPDA Tracking}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item Tracks scatterers in 3D spatial domain (not Range-Doppler).
        \item JPDA handles data association; EKF filters state with covariance.
        \item Output: Clean 3D point cloud with ghosts removed.
    \end{itemize}
    \begin{minipage}{0.48\linewidth}
    \centering
    \posterfig{\linewidth}{fig_radar_vs_camera_p3_0053.png}{(a) Before removal: ghost targets remain}
    \end{minipage}
    \hfill
    \begin{minipage}{0.48\linewidth}
    \posterfig{\linewidth}{fig_radar_vs_camera_p5_0065.png}{(b) After removal: ghosts rejected}
    \end{minipage}
    \end{minipage}

% COLUMN 3: PCTtrans + MAML + RESULTS

\subcolumn{0.5}
    \block{\textcolor{white}{\large \ding{208}\ \ Few-Shot PID with PCTtrans}}{
    \small
    \posterfig{0.95\linewidth}{fig_pcttrans_architecture.png}{PCTtrans: frame-wise PCT $+$ Temporal Transformer}
    \begin{minipage}{\linewidth}
    \textbf{Preprocessing:}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item DBSCAN ($k=8$, $S=1.0$) + near-field filter
        \item Centroid normalization: $p_i' = p_i - (\bar{x}, \bar{y}, 0)$
        \item Random padding to $n_{\max}=1000$
        \item Output: $\mathcal{S} = \{\mathcal{P}_t\}_{t=1}^{10}$, $\mathcal{P}_t \in \mathbb{R}^{1000 \times 5}$
    \end{itemize}
    \textbf{Spatial (PCT Encoder):}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item SG1: FPS $n_1=200$, $r_1=0.2$; SG2: FPS $n_2=100$, $r_2=0.3$
        \item Neighbor Embedding: $\Delta F(p) = \text{concat}_{q \in \text{knn}(p,k)} (F(q) - F(p))$
        \item Offset-Attention: $F_{\text{out}} = \text{LBR}(F_{\text{in}} - F_{\text{sa}}) + F_{\text{in}}$
        \item Output: $\mathbf{f}_t \in \mathbb{R}^{192}$
    \end{itemize}
    \textbf{Temporal (Transformer):}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item Positional encoding: $\mathbf{f}_t^{\text{pos}} = \mathbf{f}_t + \text{PE}_t$
        \item Multi-head self-attention ($h=4$)
        \item Max + average pooling $\to$ $\mathbf{g} \in \mathbb{R}^{192}$
        \item Classification:
        $$p(y|\mathcal{S}) = \text{Softmax}(W_2 \cdot \text{ReLU}(W_1 \mathbf{g} + b_1) + b_2)$$
    \end{itemize}
    \textbf{Key Design Choices:}
    \begin{itemize}
        \setlength\itemsep{0.5mm}
        \item FPS + Ball Query handles non-uniform density.
        \item Offset-Attention emphasizes local structure.
        \item 4-head attention captures gait dynamics.
    \end{itemize}
    \end{minipage}
    }
    \block{\textcolor{white}{\large \ding{209}\ \ Pre-trained Meta-Learning}}{
    \small
    \textbf{Two-Phase Algorithm:}
    \vspace{1mm}
    \colorbox{deepLearning!12}{\parbox{0.95\linewidth}{\textbf{Phase 1:} Supervised training on 3 subjects $\to \theta_0^{PT}$}}
    \vspace{1mm}
    \colorbox{deepLearning!12}{\parbox{0.95\linewidth}{\textbf{Phase 2:} MAML meta-training with $\theta_0^{PT}$}}
    \vspace{2mm}
    Inner loop: $\theta_i' = \theta - \alpha\nabla\mathcal{L}$ ($\alpha{=}0.1$). Outer loop: $\theta \leftarrow \theta - \beta\nabla\sum_i\mathcal{L}$ ($\beta{=}0.001$).
    \vspace{2mm}
    \centering
    \colorbox{resultsColor!12}{\parbox{0.95\linewidth}{\centering
    \textbf{Gain:} +15.6pp over random MAML
    }}
    }
